#!/bin/sh
# tx_watchdog.sh disabled 2026-09-14.
#
# It polled the TX buffer and applied a fixed -10 dB of attenuation a couple
# of seconds after any stream started. That was a workaround for the kernel
# unmute overwriting a gain set before streaming; firmware patches/0005 fixes
# that properly, so the gain you set is now the gain you get, and starting a
# stream having set nothing still restores your last one.
#
# Re-enable by uncommenting the line below. The original file is autorun.sh.orig
# and the script itself is still at /mnt/jffs2/tx_watchdog.sh.
#
# /mnt/jffs2/tx_watchdog.sh &

# --- until the rootfs carries firmware/patches/0013 --------------------------
# Do here what the patched S40network will do natively. hwaddress goes on BOTH
# branches: the macb driver finds no MAC in the device tree and randomises one
# every boot whether the address is static or from DHCP. hostname goes only on
# the dhcp branch, because DHCP option 12 is what a router lists you by and a
# static address never sends one.
ETHADDR=$(fw_printenv -n ethaddr 2>/dev/null)
HN=$(hostname)
if [ -n "$ETHADDR" ] && ! grep -q hwaddress /etc/network/interfaces; then
    awk -v mac="$ETHADDR" -v hn="$HN" '
        {print}
        /^iface eth0 inet dhcp$/  {print "\thostname " hn; print "\thwaddress ether " mac}
        /^iface eth0 inet static$/ {print "\thwaddress ether " mac}
    ' /etc/network/interfaces > /tmp/if.new && cp /tmp/if.new /etc/network/interfaces
    ifdown eth0 2>/dev/null
    ifup eth0 2>/dev/null
    sleep 6
    ip -4 addr show eth0 | grep -q "inet " || udhcpc -i eth0 -n -q 2>/dev/null || true
fi
